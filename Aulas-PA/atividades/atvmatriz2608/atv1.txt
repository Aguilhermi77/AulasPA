import javax.swing.JOptionPane;
void main() {
//declaraçao matriz A
    int[][] A = new int[3][3];
    //declaraçao matriz B
    int[][] B = new int[3][3];

    //  Preenchendo a Matriz A
    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            String entrada = JOptionPane.showInputDialog(
                    null,
                    "Matriz A - Digite o valor para a posição [" + i + "][" + j + "]:");
            A[i][j] = Integer.parseInt(entrada);
        }
    }

    //  Preenchendo a Matriz B
    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            String entrada = JOptionPane.showInputDialog(
                    null,
                    "Matriz B - Digite o valor para a posição [" + i + "][" + j + "]:");
            B[i][j] = Integer.parseInt(entrada);
        }
    }

    //declaraçao matriz C
    int[][] C = new int[3][3];
    for(int i=0; i <3; i++){
        for(int j=0; j <3; j++){
            C[i][j] = A[i][j] + B[i][j];
        }
    }
    //declaraçao matriz D
    int[][] D = new int[3][3];
    for(int i=0; i <3; i++){
        for(int j=0; j <3; j++){
            D[i][j] = A[i][j] - B[i][j];
        }
    }
    //declaraçao matriz E
    int[][] E = new int[3][3];
    for(int i=0; i <3; i++){
        for(int j=0; j <3; j++){
            E[i][j] = A[i][j] / B[i][j];
        }
    }
    //exibe matriz A
    System.out.println("Matriz A:");
    for (int i = 0; i<3; i++){
        for (int j = 0; j<3; j++){
            System.out.print(A[i][j] +"\t");
        }
        System.out.println();
    }
    //exibe matriz B
    System.out.println("Matriz B:");
    for (int i = 0; i<3; i++){
        for (int j = 0; j<3; j++){
            System.out.print(B[i][j] +"\t");
        }
        System.out.println();
    }
    //exibe matriz C
    System.out.println("Matriz C(soma):");
    for (int i = 0; i<3; i++){
        for (int j = 0; j<3; j++){
            System.out.print(C[i][j] +"\t");
        }
        System.out.println();
    }
    //exibe matriz D
    System.out.println("Matriz C(subtraçao):");
    for (int i = 0; i<3; i++){
        for (int j = 0; j<3; j++){
            System.out.print(D[i][j] +"\t");
        }
        System.out.println();
    }
    //exibe matriz E
    System.out.println("Matriz C(divisao):");
    for (int i = 0; i<3; i++){
        for (int j = 0; j<3; j++){
            System.out.print(E[i][j] +"\t");
        }
        System.out.println();
    }
}
